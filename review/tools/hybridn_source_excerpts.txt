42: def blend_weight(envelope_db: np.ndarray, config: CrossoverConfig) -> np.ndarray:
43:     """Compute the Amp B mix weight `t` in [0, 1] for each sample of envelope_db.

48:     start = config.crossover_dbfs - width / 2.0

54: def blend(
55:     dry_envelope_db: np.ndarray,
56:     amp_a_render: np.ndarray,
57:     amp_b_render: np.ndarray,
58:     config: CrossoverConfig,
59: ) -> tuple[np.ndarray, np.ndarray]:
60:     """Blend two (already aligned, already trimmed-if-desired) amp renders.
61: 
62:     Uses a LINEAR amplitude crossfade (not equal-power) because amp_a_render and
63:     amp_b_render are correlated versions of the same guitar signal, not
64:     independent sources -- equal-power crossfades assume decorrelated content
65:     and would produce a level bump in the transition region here.
66: 
67:     Returns (hybrid_audio, t_curve) where t_curve is the per-sample Amp B weight
68:     actually used, useful for debugging/plotting the transition.
69:     """
70:     n = min(len(dry_envelope_db), len(amp_a_render), len(amp_b_render))
71:     envelope_db = dry_envelope_db[:n]
72:     a = amp_a_render[:n] * (10.0 ** (config.amp_a_trim_db / 20.0))
73:     b = amp_b_render[:n] * (10.0 ** (config.amp_b_trim_db / 20.0))
74: 
75:     t = blend_weight(envelope_db, config)
76:     hybrid = a * (1.0 - t) + b * t
77:     return hybrid, t

hybrid/core/envelope.py
168: def bounded_causal_envelope_db(
169:     audio: np.ndarray,
170:     sample_rate: int,
171:     config: BoundedEnvelopeConfig = DEFAULT_BOUNDED_ENVELOPE_CONFIG,
172: ) -> np.ndarray:
173:     """The PRODUCTION crossover envelope: strictly causal AND strictly
174:     bounded-memory (see module docstring for why `rms_envelope_db`'s one-pole
175:     release is not acceptable for this).
176: 
177:     Three cascaded FIR (non-recursive) stages, each a fixed causal window:
178: 
179:       1. Causal moving RMS over `rms_window_ms` (dB), exactly like
180:          `rms_envelope_db`'s first stage.
181:       2. Causal moving average over `attack_avg_ms` of stage 1's dB values --
182:          a bounded stand-in for the old one-pole attack smoothing, replacing
183:          "abrupt" transient jumps with a short, finite-length blur instead of
184:          instant response.
185:       3. Causal decaying-maximum ("peak hold with linear dB decay") over
186:          `release_window_ms`: sample i is
187:          `max_{k in [i-W+1, i]} (stage2_db[k] - release_range_db * (i-k) / W)`.
188:          This is the bounded replacement for a one-pole release: a loud peak's
189:          influence provably reaches exactly zero `release_window_ms` later,
190:          not asymptotically. Implemented as a shifted causal sliding-max (see
191:          `_causal_sliding_max`), which is the standard O(n) technique for a
192:          decaying peak-hold detector.
193: 
194:     Deterministic, causal (no sample depends on any `audio[j]` for `j > i`),
195:     and its total dependency on `audio` is bounded by
196:     `bounded_envelope_max_history_samples(sample_rate, config)` -- see
197:     tests/test_envelope_finite.py.
198:     """
199:     audio = np.asarray(audio, dtype=np.float64)
200:     n = len(audio)
201:     if n == 0:
202:         return np.zeros(0, dtype=np.float64)
203: 
204:     w1 = _ms_to_samples(config.rms_window_ms, sample_rate)
205:     squared = audio * audio
206:     padded = np.concatenate([np.zeros(w1 - 1, dtype=np.float64), squared])
207:     cumsum = np.concatenate([[0.0], np.cumsum(padded)])
208:     window_sum = cumsum[w1:] - cumsum[:n]
209:     rms = np.sqrt(np.maximum(window_sum / w1, 0.0))
210:     stage1_db = 20.0 * np.log10(np.maximum(rms, _EPS))
211: 
212:     w2 = _ms_to_samples(config.attack_avg_ms, sample_rate)
213:     stage2_db = _causal_moving_mean(stage1_db, w2)
214: 
215:     w3 = _ms_to_samples(config.release_window_ms, sample_rate)
216:     if w3 <= 1:
217:         return stage2_db
218: 
219:     decay_per_sample = config.release_range_db / w3
220:     idx = np.arange(n, dtype=np.float64)
221:     ramped = stage2_db + decay_per_sample * idx
222:     floor_db = float(stage2_db.min()) - config.release_range_db - 1.0
223:     floor_ramped = floor_db  # sentinel is below any real ramped value at idx=0
224:     windowed_max = _causal_sliding_max(ramped, w3, floor=floor_ramped)
225:     stage3_db = windowed_max - decay_per_sample * idx
226:     return stage3_db

1005: // --- "Guitar volume feel" 0-10 knob -----------------------------------
1006: // A cosmetic, per-DI-calibrated alternative to the raw dBFS crossover
1007: // slider (see static/app.js's crossoverSlider above, which remains the
1008: // single source of truth -- this knob only ever reads/writes that value).
1009: // crossoverKnobCalibration.{minDb,maxDb} anchor "0"/"10" to this render's
1010: // OWN measured instrument-level-into-NAM envelope (blend_envelope_percentiles
1011: // from /api/render_pair, i.e. pair.envelope_db -- the profiled envelope
1012: // that actually drives build_hybrid()'s crossfade, per the user's request
1013: // to calibrate against real "instrument level into NAM" signal rather than
1014: // an assumed universal volume-pot law -- see docs/README known caveats on
1015: // why a literal per-guitar pot-taper simulation is deliberately not done).
1016: // Mapping the knob LINEARLY onto that dBFS range is already the "audio
1017: // taper" behaviour: dB is itself a log scale of amplitude, so a linear
1018: // sweep in dB reads as the perceptually-log sweep a real volume pot aims
1019: // for -- no extra exponential curve is needed on top.
1020: const crossoverKnobSlider = document.getElementById("crossover-knob-slider");
1021: const crossoverKnobValue = document.getElementById("crossover-knob-value");
1022: const crossoverKnobNote = document.getElementById("crossover-knob-note");
1023: let crossoverKnobCalibration = { minDb: -40.0, maxDb: -6.0, calibrated: false };
1024: 
1025: function dbFromKnob(knobValue) {
1026:   const { minDb, maxDb } = crossoverKnobCalibration;
1027:   return minDb + (parseFloat(knobValue) / 10.0) * (maxDb - minDb);
1028: }
1029: 
1030: function knobFromDb(db) {
1031:   const { minDb, maxDb } = crossoverKnobCalibration;
1032:   if (maxDb <= minDb) return 0;
1033:   return Math.min(10, Math.max(0, ((parseFloat(db) - minDb) / (maxDb - minDb)) * 10.0));
1034: }
